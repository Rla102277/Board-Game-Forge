    54	function useWorkspacesList() {
    55	  const [workspaces, setWorkspaces] = useState<Workspace[] | null>(null);
    56	  const [error, setError] = useState<string | null>(null);
    57	  const reload = async () => {
    58	    try {
    59	      const list = await workspacesApi.list();
    60	      setWorkspaces(list);
    61	    } catch (err) {
    62	      setError(err instanceof Error ? err.message : String(err));
    63	    }
    64	  };
    65	  useEffect(() => { reload(); }, []);
    66	  return { workspaces, error, reload };
    67	}

   274	              What game do you want to make today?
   275	            </h1>
   276	            <p className="text-muted-foreground text-lg">
   277	              Describe it, or pick a template. We'll spin up a playable prototype with rules, components, and player roles in seconds.
   278	            </p>
   279	          </div>
   280	
   281	          <div className="rounded-2xl border bg-card p-4 shadow-lg">
   282	            <Textarea
   283	              data-testid="generate-prompt"
   284	              value={prompt}
   285	              onChange={(e) => setPrompt(e.target.value)}
   286	              placeholder="A 2-4 player cooperative game where you're space cartographers mapping a fragile alien sector before it collapses. Push your luck, share resources, win together."

   295	                      <Plus className="h-4 w-4" /> Empty project

   314	              </div>
   315	              <Button

   317	                className="gap-2"
   318	                onClick={() => generateFrom(null)}
   319	                disabled={!prompt.trim() || Boolean(generating)}
   320	                data-testid="generate-from-prompt"
   321	              >
   322	                {generating === "prompt" ? <Loader2 className="h-4 w-4 animate-spin" /> : <Sparkles className="h-4 w-4" />}
   323	                Generate

   325	              </Button>

   338	          </div>

   340	            {TEMPLATES.map((t) => (

   343	                onClick={() => generateFrom(t.key)}
   344	                disabled={Boolean(generating)}
   345	                data-testid={`template-${t.key}`}

   347	              >

   349	                  <div className="text-3xl">{t.emoji}</div>

   363	        {/* Learn the craft — prominent training surface */}

   537	    const arr = JSON.parse(raw);
