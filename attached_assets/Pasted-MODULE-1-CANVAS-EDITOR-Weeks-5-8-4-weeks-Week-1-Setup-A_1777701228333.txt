import { projectsTable } from './projects';

export const gameBoardsTable = pgTable('game_boards', {
  id: serial('id').primaryKey(),
  projectId: integer('project_id').references(() => projectsTable.id, { onDelete: 'cascade' }),
  name: text('name').notNull().default('Board 1'),
  width: integer('width').notNull().default(800),
  height: integer('height').notNull().default(600),
  gridSize: integer('grid_size').notNull().default(20),
  showGrid: integer('show_grid').notNull().default(1), // boolean as int
  snapToGrid: integer('snap_to_grid').notNull().default(1),
  canvasState: jsonb('canvas_state').notNull(), // Fabric.js canvas JSON
  version: integer('version').notNull().default(1),
  createdAt: timestamp('created_at', { withTimezone: true }).default(sql`now()`),
  updatedAt: timestamp('updated_at', { withTimezone: true }).default(sql`now()`),
});

export type GameBoard = typeof gameBoardsTable.$inferSelect;
export type GameBoardInsert = typeof gameBoardsTable.$inferInsert;

import { useEffect, useRef, useState } from 'react';
import { Canvas as FabricCanvas } from 'fabric';
import { Button } from '@/components/ui/button';
import { Card } from '@/components/ui/card';

interface CanvasTabProps {
  projectId: number;
}

export default function CanvasTab({ projectId }: CanvasTabProps) {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const fabricCanvasRef = useRef<FabricCanvas | null>(null);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    if (!canvasRef.current) return;

    // Initialize Fabric.js canvas
    const canvas = new FabricCanvas(canvasRef.current, {
      width: 800,
      height: 600,
      backgroundColor: '#f0f0f0',
    });

    fabricCanvasRef.current = canvas;
    setIsLoading(false);

    return () => {
      canvas.dispose();
    };
  }, []);

  const addRectangle = () => {
    const canvas = fabricCanvasRef.current;
    if (!canvas) return;

    const rect = new fabric.Rect({
      left: 100,
      top: 100,
      width: 100,
      height: 100,
      fill: '#3b82f6',
    });

    canvas.add(rect);
    canvas.renderAll();
  };

  const addCircle = () => {
    const canvas = fabricCanvasRef.current;
    if (!canvas) return;

    const circle = new fabric.Circle({
      left: 100,
      top: 100,
      radius: 50,
      fill: '#ef4444',
    });

    canvas.add(circle);
    canvas.renderAll();
  };

  const saveBoard = async () => {
    const canvas = fabricCanvasRef.current;
    if (!canvas) return;

    const canvasJSON = canvas.toJSON();

    try {
      const response = await fetch(`/api/projects/${projectId}/boards`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          name: 'Board 1',
          canvasState: canvasJSON,
        }),
      });

      if (response.ok) {
        console.log('Board saved successfully');
      }
    } catch (error) {
      console.error('Failed to save board:', error);
    }
  };

  if (isLoading) return <div>Loading canvas...</div>;

  return (
    <div className="w-full h-full flex flex-col gap-4 p-4">
      <div className="flex gap-2">
        <Button onClick={addRectangle}>Add Rectangle</Button>
        <Button onClick={addCircle}>Add Circle</Button>
        <Button onClick={saveBoard} variant="default">
          Save Board
        </Button>
      </div>
      <Card>
        <canvas ref={canvasRef} className="border border-slate-300" />
      </Card>
    </div>
  );
}

  try {
    const { projectId, boardId } = req.params;
    const [board] = await db

      .from(gameBoardsTable)
      .where(
        and(
          eq(gameBoardsTable.id, parseInt(boardId)),
          eq(gameBoardsTable.projectId, parseInt(projectId))

        name: name || 'New Board',
        width: width || 800,
        height: height || 600,
        canvasState: canvasState || {},

  } catch (error) {
    res.status(500).json({ error: 'Failed to create board' });

  try {
    const { projectId, boardId } = req.params;
    const { name, canvasState } = req.body;

    const [updated] = await db
      .update(gameBoardsTable)
      .set({
        name: name,
        canvasState: canvasState,
        updatedAt: new Date(),

      .where(
        and(
          eq(gameBoardsTable.id, parseInt(boardId)),

const AUTO_SAVE_INTERVAL = 30000; // 30 seconds

export function setupAutoSave(
  canvas: fabric.Canvas,
  projectId: number,
  boardId: number,
  onSave?: () => void
) {
  const interval = setInterval(async () => {
    const canvasJSON = canvas.toJSON();
    
    try {
      await fetch(`/api/projects/${projectId}/boards/${boardId}`, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ canvasState: canvasJSON }),
      });
      onSave?.();
    } catch (error) {
      console.error('Auto-save failed:', error);
    }
  }, AUTO_SAVE_INTERVAL);

  return () => clearInterval(interval);
}

import { Router } from 'express';
import { db, playtestSessionsTable, projectsTable } from '@workspace/db';
import { eq } from 'drizzle-orm';
import { sendPlaytestInvite, sendPlaytestReminder } from '../lib/email-service';

const router = Router();

// Schedule a playtest session
router.post('/projects/:projectId/playtest-sessions/:sessionId/schedule', async (req, res) => {
  try {
    const { projectId, sessionId } = req.params;
    const { scheduledDate, scheduledTime, inviteEmails } = req.body;

    // Get project for game title
    const [project] = await db.select().from(projectsTable).where(eq(projectsTable.id, parseInt(projectId)));
    if (!project) return res.status(404).json({ error: 'Project not found' });

    // Update session
    const [updated] = await db
      .update(playtestSessionsTable)
      .set({
        scheduledDate: new Date(scheduledDate),
        scheduledTime: scheduledTime,
        inviteEmails: inviteEmails,
        status: 'scheduled',
      })
      .where(eq(playtestSessionsTable.id, parseInt(sessionId)))
      .returning();

    // Send invites
    const feedbackLink = `${process.env.VITE_APP_URL}/feedback/${sessionId}`;
    await sendPlaytestInvite(project.name, scheduledDate, scheduledTime, feedbackLink, inviteEmails);

    res.json({ success: true, session: updated });
  } catch (error) {
    console.error(error);
    res.status(500).json({ error: 'Failed to schedule playtest' });
  }
});

// Get upcoming playtest sessions
router.get('/projects/:projectId/playtest-sessions/upcoming', async (req, res) => {
  try {
    const projectId = parseInt(req.params.projectId);
    const now = new Date();

    const sessions = await db
      .select()
      .from(playtestSessionsTable)
      .where(
        and(
          eq(playtestSessionsTable.projectId, projectId),
          eq(playtestSessionsTable.status, 'scheduled'),
          gt(playtestSessionsTable.scheduledDate, now)
        )
      )
      .orderBy(playtestSessionsTable.scheduledDate);

    res.json(sessions);
  } catch (error) {
    res.status(500).json({ error: 'Failed to fetch sessions' });
  }
});

export default router;

export function PlaytestScheduler({
  projectId,
  sessionId,
  gameTitle,
}: PlaytestSchedulerProps) {
  const [selectedDate, setSelectedDate] = useState<Date | undefined>();
  const [selectedTime, setSelectedTime] = useState('14:00');
  const [inviteEmails, setInviteEmails] = useState('');

        }
      );

        alert('Playtest scheduled and invites sent!');
        setSelectedDate(undefined);
        setSelectedTime('14:00');
        setInviteEmails('');
      }
    } catch (error) {
      console.error('Failed to schedule:', error);
    } finally {
      setIsScheduling(false);
    }
  };

  return (
    <Card className="p-6 max-w-md">
      <h3 className="text-lg font-semibold mb-4">Schedule Playtest: {gameTitle}</h3>

      <div className="mb-4">
        <label className="block text-sm font-medium mb-2">Date</label>

          onSelect={setSelectedDate}

        />
      </div>

      <div className="mb-4">

          value={selectedTime}

import puppeteer from 'puppeteer';
import * as fs from 'fs';
import * as path from 'path';

export async function generatePDF(
  htmlContent: string,
  filename: string,
  options?: {
    format?: string;
    margin?: { top: string; right: string; bottom: string; left: string };
  }
): Promise<Buffer> {
  const browser = await puppeteer.launch();
  const page = await browser.newPage();

  await page.setContent(htmlContent, { waitUntil: 'networkidle0' });

  const pdf = await page.pdf({
    format: options?.format || 'A4',
    margin: options?.margin || {
      top: '1cm',
      right: '1cm',
      bottom: '1cm',
      left: '1cm',
    },
  });

  await browser.close();
  return pdf;
}

export async function generateRulebookPDF(
  gameTitle: string,
  rulebookHtml: string
): Promise<Buffer> {
  const wrappedHtml = `
    <!DOCTYPE html>
    <html>
    <head>
      <meta charset="UTF-8">
      <style>
        body { font-family: Arial, sans-serif; line-height: 1.6; color: #333; }
        h1 { color: #1a1a1a; border-bottom: 3px solid #000; padding-bottom: 10px; }
        h2 { color: #444; margin-top: 20px; }
        .page-break { page-break-after: always; }
        table { width: 100%; border-collapse: collapse; margin: 10px 0; }
        th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
        th { background-color: #f0f0f0; }
      </style>
    </head>
    <body>
      <h1>${gameTitle} — Rulebook</h1>
      <div class="page-break"></div>
      ${rulebookHtml}
      <p style="margin-top: 40px; color: #999; font-size: 0.9em;">
        Generated by Table Forge on ${new Date().toLocaleDateString()}
      </p>
    </body>
    </html>
  `;

  return generatePDF(wrappedHtml, `${gameTitle}-rulebook.pdf`);
}

      const data = await response.json();
      setRulebookHtml(data.html);
    } catch (error) {
      console.error('Failed to fetch rulebook:', error);
    } finally {
      setIsLoading(false);
    }
  };

  const downloadPDF = async () => {
    setIsDownloading(true);
    try {
      const response = await fetch(`/api/projects/${projectId}/rulebook/download`);
      const blob = await response.blob();
      const url = window.URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = 'rulebook.pdf';
      a.click();
    } catch (error) {
      console.error('Failed to download PDF:', error);
    } finally {
      setIsDownloading(false);
    }
  };

  if (isLoading) return <div>Loading rulebook...</div>;

  return (
    <div className="w-full">
      <div className="flex gap-2 mb-4">
        <Button onClick={downloadPDF} disabled={isDownloading}>
          <Download className="w-4 h-4 mr-2" />
          {isDownloading ? 'Generating PDF...' : 'Download PDF'}
        </Button>
        <Button variant="outline" onClick={fetchRulebook}>
          <Eye className="w-4 h-4 mr-2" /> Refresh
        </Button>
      </div>

      <Card className="p-6 bg-slate-900 prose prose-invert max-w-none">
        <div dangerouslySetInnerHTML={{ __html: rulebookHtml }} />
      </Card>
    </div>
  );
}

        playerCount: projectsTable.playerCount,
        targetDuration: projectsTable.targetDuration,
        rating: gameLibraryTable.averageRating,
        slug: gameLibraryTable.slug,
      })
      .from(gameLibraryTable)
      .innerJoin(projectsTable, eq(gameLibraryTable.projectId, projectsTable.id))
      .where(eq(gameLibraryTable.isPublished, 1));

    // Add filters
    if (search) {
      // SQL LIKE search
    }

    // Sort options
    if (sort === 'rating') {
      query = query.orderBy(desc(gameLibraryTable.averageRating));
    } else if (sort === 'newest') {
      query = query.orderBy(desc(gameLibraryTable.publishedAt));
    } else if (sort === 'popular') {
      query = query.orderBy(desc(gameLibraryTable.totalPlays));

// GET single game (public)
router.get('/library/games/:slug', async (req, res) => {
  try {
    const gameSlug = req.params.slug;

    const [game] = await db
      .select()
      .from(gameLibraryTable)
      .where(eq(gameLibraryTable.slug, gameSlug));

    if (!game || !game.isPublished) {
      return res.status(404).json({ error: 'Game not found' });
    }

    const project = await db
      .select()
      .from(projectsTable)
      .where(eq(projectsTable.id, game.projectId));

    const ratings = await db
      .select()
      .from(gameRatingsTable)
      .where(eq(gameRatingsTable.gameLibraryId, game.id));

    res.json({ ...game, ...project[0], ratings });
  } catch (error) {
    res.status(500).json({ error: 'Failed to fetch game' });
  }
});

// PUBLISH a game (authenticated)
router.post('/projects/:projectId/publish', async (req, res) => {
  try {
    const projectId = parseInt(req.params.projectId);

    const [project] = await db
      .select()
      .from(projectsTable)
      .where(eq(projectsTable.id, projectId));

    if (!project) return res.status(404).json({ error: 'Project not found' });

    const gameSlug = slug(project.name).toLowerCase();

    const [published] = await db
      .insert(gameLibraryTable)
      .values({
        projectId,
        isPublished: 1,
        slug: gameSlug,
        publishedAt: new Date(),
      })
      .returning();

    res.json({ success: true, published });
  } catch (error) {
    res.status(500).json({ error: 'Failed to publish game' });
  }
});

// SUBMIT rating (public)
router.post('/library/games/:gameLibraryId/ratings', async (req, res) => {
  try {
    const { gameLibraryId } = req.params;
    const { userId, rating, review } = req.body;

    const [submitted] = await db
      .insert(gameRatingsTable)
      .values({
        gameLibraryId: parseInt(gameLibraryId),
        userId,
        rating,
        review,
      })
      .returning();

    // Update average rating
    const allRatings = await db
      .select()
      .from(gameRatingsTable)
      .where(eq(gameRatingsTable.gameLibraryId, parseInt(gameLibraryId)));

    const avgRating = (
      allRatings.reduce((sum, r) => sum + (r.rating || 0), 0) / allRatings.length
    ).toFixed(2);

    await db
      .update(gameLibraryTable)
      .set({
        averageRating: parseFloat(avgRating),
        totalRatings: allRatings.length,
      })
      .where(eq(gameLibraryTable.id, parseInt(gameLibraryId)));

    res.json({ success: true, rating: submitted });
  } catch (error) {
    res.status(500).json({ error: 'Failed to submit rating' });
  }
});
