ts// artifacts/api-server/src/routes/analysis.ts - new endpoint
router.post(
  "/projects/:projectId/design-advisor",
  async (req, res): Promise<void> => {
    // 1. Fetch all project data in parallel
    const [project, rules, entities, players, feedback, balance, changelog] = await Promise.all([...]);
    
    // 2. Build comprehensive game description
    const gameContext = `
      Project: ${project.name}
      Phase: ${project.designPhase}
      Rules: ${rules.map(r => `[${r.maturity}] ${r.title}: ${r.content}`).join('\n')}
      Components: ${entities.map(e => `${e.name} (${e.type})`).join(', ')}
      Players: ${players.map(p => `${p.name} (${p.role})`).join(', ')}
      Recent Feedback: ${feedback.map(f => f.summary).join('; ')}
      Balance Score: ${balance.score}
      Experimental Rules: ${rules.filter(r => r.maturity === 'experimental').length}
    `;
    
    // 3. Send to Claude with structured prompt
    const prompt = `You are a board game design consultant. Analyze this game design:
    ${gameContext}
    
    Provide analysis in JSON:
    {
      "strengths": ["...", "..."],
      "issues": [
        {
          "title": "...",
          "description": "...",
          "priority": "high|medium|low",
          "estimatedHours": number
        }
      ],
      "recommendations": [
        {
          "action": "...",
          "why": "...",
          "hoursNeeded": number,
          "priority": 1|2|3|...
        }
      ],
      "nextMilestone": {
        "name": "Move to Beta",
        "requirements": ["...", "..."],
        "progressPercentage": number
      }
    }`;
    
    // 4. Return structured response
  }
);
